Đang tải...
SQL cần chạy 1 lần trên Supabase 
create table if not exists game_announcements (
  id text primary key,
  title text not null,
  body text not null,
  date_label text,
  active boolean default true,
  created_at timestamptz default now()
);
alter table game_announcements enable row level security;
create policy "ann_select" on game_announcements for select to anon, authenticated using (true);
create policy "ann_insert" on game_announcements for insert to anon, authenticated with check (true);
create policy "ann_update" on game_announcements for update to anon, authenticated using (true) with check (true);
create policy "ann_delete" on game_announcements for delete to anon, authenticated using (true);

SQL Editor → dán → Run. Xong là đăng được.
